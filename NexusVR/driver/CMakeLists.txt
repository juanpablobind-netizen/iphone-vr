cmake_minimum_required(VERSION 3.20)
project(NexusHands LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 required")
endif()
if(NOT EXISTS "${OPENVR_ROOT}/headers/openvr_driver.h")
  message(FATAL_ERROR "Set OPENVR_ROOT to the Valve OpenVR SDK")
endif()
add_library(driver_nexushands SHARED driver.cpp)
target_compile_features(driver_nexushands PRIVATE cxx_std_17)
target_include_directories(driver_nexushands PRIVATE "${OPENVR_ROOT}/headers")
target_link_libraries(driver_nexushands PRIVATE ws2_32)
set_target_properties(driver_nexushands PROPERTIES PREFIX "")
install(TARGETS driver_nexushands RUNTIME DESTINATION bin/win64 LIBRARY DESTINATION bin/win64)
install(FILES driver.vrdrivermanifest DESTINATION .)
install(DIRECTORY resources DESTINATION .)
